// ادمج هذه الأقسام داخل: match /databases/{database}/documents { ... }
// اترك قواعد users و usernames كما هي، واستبدل قواعد invites و links بما يلي، ثم اختبرها في Rules Playground.

function linkPath(uid, t) { return /databases/$(database)/documents/links/$(uid + '_' + t); }

match /invites/{code} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
  // صاحب الكود يحذفه، والمعلّم يحذفه فقط في نفس العملية التي تنشئ/تُعيد الربط (استعمال لمرة واحدة)
  allow delete: if request.auth != null && (
    resource.data.uid == request.auth.uid ||
    (existsAfter(linkPath(resource.data.uid, request.auth.uid)) &&
     getAfter(linkPath(resource.data.uid, request.auth.uid)).data.code == code &&
     getAfter(linkPath(resource.data.uid, request.auth.uid)).data.t == request.auth.uid));
}

match /links/{id} {
  function mem(d) { return request.auth != null && (request.auth.uid == d.s || request.auth.uid == d.t); }
  function ks() { return request.resource.data.diff(resource.data).affectedKeys(); }
  function inv(c) { return /databases/$(database)/documents/invites/$(c); }

  allow read: if resource == null || mem(resource.data);
  allow delete: if mem(resource.data);
  allow create: if request.auth != null
    && request.resource.data.t == request.auth.uid
    && id == request.resource.data.s + '_' + request.auth.uid
    && get(inv(request.resource.data.code)).data.uid == request.resource.data.s;

  allow update: if mem(resource.data) && (
    // الطالب: sum, dn, sn, ويستطيع فقط تحويل off إلى true
    (request.auth.uid == resource.data.s
      && ks().hasOnly(['sum','dn','sn','off'])
      && (request.resource.data.get('off', false) == true
          || request.resource.data.get('off', false) == resource.data.get('off', false)))
    ||
    // المعلّم: الواجبات والاسم
    (request.auth.uid == resource.data.t && ks().hasOnly(['as','tn']))
    ||
    // المعلّم: فصل الطالب
    (request.auth.uid == resource.data.t && ks().hasOnly(['off']) && request.resource.data.off == true)
    ||
    // المعلّم: إعادة الربط، لا تتم إلا باستهلاك كود جديد للطالب
    (request.auth.uid == resource.data.t
      && resource.data.get('off', false) == true && request.resource.data.off == false
      && ks().hasOnly(['off','code','tn'])
      && get(inv(request.resource.data.code)).data.uid == resource.data.s
      && !existsAfter(inv(request.resource.data.code)))
  );
}

// صور الواجبات
match /links/{id}/files/{fid} {
  function mem() {
    let l = get(/databases/$(database)/documents/links/$(id)).data;
    return request.auth != null && (request.auth.uid == l.s || request.auth.uid == l.t);
  }
  allow read, delete: if mem();
  allow create: if mem() && request.resource.data.u == request.auth.uid && request.resource.data.d.size() < 800000;
}
